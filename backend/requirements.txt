# UniFlow Backend — Python dependencies
# Python 3.12
#
# Setup:
#   python -m venv venv
#   source venv/bin/activate        # Windows: venv\Scripts\activate
#   pip install -r requirements.txt
#
# Then create a .env (see below) and run:
#   alembic upgrade head
#   uvicorn main:app --reload --port 8000
#
# Required .env variables:
#   DATABASE_URL=postgresql://user:password@localhost:5432/pfe_backend_db
#   SECRET_KEY=<a long random string>
#   EMAIL_USER=<gmail address>            # only if using the email features
#   EMAIL_PASSWORD=<gmail app password>   # only if using the email features
#   FRONTEND_URL=http://localhost:3000
# ─────────────────────────────────────────────────────────────
